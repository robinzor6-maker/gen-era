# المرحلة 1: إزالة Legacy Validation

## ابحث عن:

```ts
if (!email)
if (!password)
if (!name)
if (!items.length)
if (password.length < ...)
```

داخل:

```text
auth.ts
orders.ts
admin.ts
products.ts
```

---

## احذف فقط ما أصبح مغطى بواسطة Zod

مثال:

### قبل

```ts
if (!email || !password) {
  return res.status(400).json({
    success: false,
    message: "Email and password required",
  });
}
```

### بعد

```ts
// removed
// handled by Zod
```

# المرحلة 2: Standard API Responses

أنشئ:

```text
src/lib/apiResponse.ts
```

```ts
export const successResponse = (
  data: unknown,
  message = "Success"
) => ({
  success: true,
  message,
  data,
});

export const errorResponse = (
  message: string,
  details?: unknown
) => ({
  success: false,
  message,
  details,
});
```

---

## بدلاً من:

```ts
res.json(user);
```

استخدم:

```ts
res.json(
  successResponse(user)
);
```

# المرحلة 3: Global Error Handler

أنشئ:

```text
src/middleware/errorHandler.ts
```

```ts
import { Request, Response, NextFunction } from "express";

export function errorHandler(
  err: Error,
  req: Request,
  res: Response,
  next: NextFunction
) {
  console.error(err);

  res.status(500).json({
    success: false,
    message: "Internal Server Error",
  });
}
```

---

## في app.ts

آخر middleware:

```ts
app.use(errorHandler);
```

# المرحلة 4: Async Wrapper

أنشئ:

```text
src/lib/asyncHandler.ts
```

```ts
import { RequestHandler } from "express";

export const asyncHandler =
  (fn: RequestHandler): RequestHandler =>
  (req, res, next) =>
    Promise.resolve(fn(req, res, next))
      .catch(next);
```

---

### قبل

```ts
router.post("/", async (req, res) => {
 ...
});
```

### بعد

```ts
router.post(
  "/",
  asyncHandler(async (req, res) => {
    ...
  })
);
```
