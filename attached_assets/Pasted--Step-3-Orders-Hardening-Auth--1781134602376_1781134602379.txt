# 🧠 2. ORDER SERVICE (CORE LOGIC)

## 📁 جديد:

`services/orderService.ts`

```ts id="order_service"
import { db } from "@workspace/db";
import { orders } from "@workspace/db/schema/orders";
import { orderItems } from "@workspace/db/schema/order_items";
import { products } from "@workspace/db/schema/products";
import { eq } from "drizzle-orm";
import crypto from "crypto";

export class OrderService {

  static async createOrder(data: {
    userId?: string;
    email: string;
    items: { productId: string; quantity: number }[];
    idempotencyKey: string;
  }) {

    // 1. Check duplicate request
    const existing = await db.query.orders.findFirst({
      where: (o, { eq }) => eq(o.idempotencyKey, data.idempotencyKey),
    });

    if (existing) return existing;

    return await db.transaction(async (tx) => {

      let total = 0;

      const orderId = crypto.randomUUID();
      const orderNumber = `GE-${Date.now().toString(36).toUpperCase()}`;

      // 2. Validate stock + calculate total
      for (const item of data.items) {

        const product = await tx.query.products.findFirst({
          where: (p, { eq }) => eq(p.id, item.productId),
        });

        if (!product) throw new Error("Product not found");

        if (product.stock < item.quantity) {
          throw new Error(`Insufficient stock for ${product.name}`);
        }

        total += product.price * item.quantity;

        // 3. Atomic stock deduction
        await tx
          .update(products)
          .set({
            stock: product.stock - item.quantity,
          })
          .where(eq(products.id, product.id));

        // 4. Insert order item
        await tx.insert(orderItems).values({
          orderId,
          productId: product.id,
          quantity: item.quantity,
          priceSnapshot: product.price,
        });
      }

      // 5. Create order
      const order = await tx.insert(orders).values({
        id: orderId,
        orderNumber,
        userId: data.userId ?? null,
        customerEmail: data.email,
        totalAmount: total,
        idempotencyKey: data.idempotencyKey,
      }).returning();

      return order[0];
    });
  }
}
```

# 🔐 3. ROUTE UPDATE

## 📁 orders.ts

```ts id="orders_route"
router.post("/", authenticate, async (req, res) => {
  try {
    const { items, email, idempotencyKey } = req.body;

    const order = await OrderService.createOrder({
      userId: req.user?.userId,
      email,
      items,
      idempotencyKey,
    });

    res.json(order);

  } catch (err: any) {
    res.status(400).json({ message: err.message });
  }
});
```
