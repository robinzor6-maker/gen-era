# 🔐 2. STRIPE SERVICE

## 📁 جديد:

`services/paymentService.ts`

```ts id="stripe_service"
import Stripe from "stripe";

const stripe = new Stripe(process.env.STRIPE_SECRET_KEY!, {
  apiVersion: "2024-06-20",
});

export class PaymentService {

  static async createPaymentIntent(amount: number, orderId: string) {
    return await stripe.paymentIntents.create({
      amount: amount * 100, // cents
      currency: "usd",
      metadata: {
        orderId,
      },
    });
  }

  static async verifyWebhook(rawBody: Buffer, signature: string) {
    return stripe.webhooks.constructEvent(
      rawBody,
      signature,
      process.env.STRIPE_WEBHOOK_SECRET!
    );
  }
}
```

## 🔥 CREATE PAYMENT INTENT

```ts id="create_intent"
router.post("/create-intent", async (req, res) => {
  const { orderId } = req.body;

  const order = await db.query.orders.findFirst({
    where: (o, { eq }) => eq(o.id, orderId),
  });

  if (!order) return res.status(404).json({ message: "Order not found" });

  const intent = await PaymentService.createPaymentIntent(
    order.totalAmount,
    order.id
  );

  await db
    .update(orders)
    .set({
      paymentIntentId: intent.id,
    })
    .where(eq(orders.id, order.id));

  res.json({
    clientSecret: intent.client_secret,
  });
});
```

# 🌐 4. WEBHOOK (CRITICAL)

## 📁 جديد:

`routes/webhooks.ts`

```ts id="webhook_route"
import express from "express";
import { PaymentService } from "../services/paymentService";
import { db } from "@workspace/db";
import { orders } from "@workspace/db/schema/orders";
import { eq } from "drizzle-orm";

const router = express.Router();

router.post("/stripe", express.raw({ type: "application/json" }), async (req, res) => {
  const signature = req.headers["stripe-signature"] as string;

  let event;

  try {
    event = PaymentService.verifyWebhook(req.body, signature);
  } catch (err) {
    return res.status(400).send("Webhook Error");
  }

  if (event.type === "payment_intent.succeeded") {
    const paymentIntent = event.data.object as any;

    const orderId = paymentIntent.metadata.orderId;

    await db
      .update(orders)
      .set({
        paymentStatus: "paid",
      })
      .where(eq(orders.id, orderId));
  }

  res.json({ received: true });
});

export default router;
```
