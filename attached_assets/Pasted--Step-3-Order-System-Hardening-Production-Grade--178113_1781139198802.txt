export const orders = pgTable("orders", {

  orderNumber: text("order_number").notNull().unique(),

  items: jsonb("items").notNull(),

  idempotencyKey: text("idempotency_key").unique(),

  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow(),
});

export function generateOrderNumber() {
  const rand = Math.random().toString(36).substring(2, 8).toUpperCase();
  const time = Date.now().toString().slice(-5);

  return `GE-${time}-${rand}`;
}

# ⚛️ 3. Atomic Order Service (Core Logic)

## 📁 `src/services/orderService.ts`

```ts
import { db } from "@workspace/db";
import { orders } from "@workspace/db/schema/orders";
import { products } from "@workspace/db/schema/products";
import { eq, sql } from "drizzle-orm";
import { generateOrderNumber } from "../utils/generateOrderNumber";

type OrderItem = {
  productId: string;
  quantity: number;
};

export async function createOrder({
  userId,
  email,
  items,
  idempotencyKey,
}: {
  userId?: string | null;
  email: string;
  items: OrderItem[];
  idempotencyKey?: string;
}) {
  // 1. منع التكرار (idempotency)
  if (idempotencyKey) {
    const existing = await db.query.orders.findFirst({
      where: (orders, { eq }) => eq(orders.idempotencyKey, idempotencyKey),
    });

    if (existing) return existing;
  }

  let total = 0;

  // 2. START TRANSACTION
  return await db.transaction(async (tx) => {
    
    const orderItems = [];

    for (const item of items) {
      const product = await tx.query.products.findFirst({
        where: (products, { eq }) => eq(products.id, item.productId),
      });

      if (!product) throw new Error("Product not found");

      // 3. CHECK STOCK
      if (product.stock < item.quantity) {
        throw new Error(`Insufficient stock for ${product.name}`);
      }

      // 4. ATOMIC STOCK DEDUCTION
      await tx
        .update(products)
        .set({
          stock: sql`${products.stock} - ${item.quantity}`,
        })
        .where(eq(products.id, product.id));

      const price = product.price * item.quantity;
      total += price;

      orderItems.push({
        productId: product.id,
        name: product.name,
        price: product.price,
        quantity: item.quantity,
      });
    }

    // 5. CREATE ORDER
    const [order] = await tx.insert(orders).values({
      orderNumber: generateOrderNumber(),
      userId: userId || null,
      customerEmail: email,
      totalAmount: total,
      items: orderItems,
      idempotencyKey: idempotencyKey || null,
    }).returning();

    return order;
  });
}
```

# 🌐 4. Orders API V2

## 📁 `src/routes/ordersV2.ts`

```ts
import { Router } from "express";
import { createOrder } from "../services/orderService";
import { authJWT } from "../middleware/authJWT";

const router = Router();

// CREATE ORDER
router.post("/", async (req, res) => {
  try {
    const { email, items, idempotencyKey } = req.body;

    const userId = req.user?.userId || null;

    const order = await createOrder({
      userId,
      email,
      items,
      idempotencyKey,
    });

    res.json(order);
  } catch (err: any) {
    res.status(400).json({ message: err.message });
  }
});

// MY ORDERS (protected)
router.get("/my", authJWT, async (req, res) => {
  const userId = req.user.userId;

  const orders = await req.db.query.orders.findMany({
    where: (orders, { eq }) => eq(orders.userId, userId),
  });

  res.json(orders);
});

export default router;
```
