# 🗄️ 1. Backend — User Dashboard APIs

## 📁 `src/routes/dashboard.ts`

```ts id="dash1"
import { Router } from "express";
import { authJWT } from "../middleware/authJWT";
import { db } from "@workspace/db";
import { orders } from "@workspace/db/schema/orders";
import { wishlist } from "@workspace/db/schema/wishlist";
import { users } from "@workspace/db/schema/users";
import { eq } from "drizzle-orm";

const router = Router();

/**
 * 👤 PROFILE
 */
router.get("/profile", authJWT, async (req: any, res) => {
  const userId = req.user.userId;

  const user = await db.query.users.findFirst({
    where: (users, { eq }) => eq(users.id, userId),
    columns: {
      password: false,
    },
  });

  res.json(user);
});

/**
 * 📦 ORDERS
 */
router.get("/orders", authJWT, async (req: any, res) => {
  const userId = req.user.userId;

  const data = await db.query.orders.findMany({
    where: (orders, { eq }) => eq(orders.userId, userId),
    orderBy: (orders, { desc }) => [desc(orders.createdAt)],
  });

  res.json(data);
});

/**
 * ❤️ WISHLIST
 */
router.get("/wishlist", authJWT, async (req: any, res) => {
  const userId = req.user.userId;

  const data = await db.query.wishlist.findMany({
    where: (wishlist, { eq }) => eq(wishlist.userId, userId),
  });

  res.json(data);
});

export default router;
```

# 🧿 2. Collectibles System (NFC / Characters Foundation)

## 📁 `schema/collectibles.ts`

```ts id="dash2"
import { pgTable, uuid, text, timestamp, integer } from "drizzle-orm/pg-core";
import { users } from "./users";
import { characters } from "./characters";

export const collectibles = pgTable("collectibles", {
  id: uuid("id").defaultRandom().primaryKey(),

  userId: uuid("user_id")
    .references(() => users.id, { onDelete: "cascade" })
    .notNull(),

  characterId: uuid("character_id")
    .references(() => characters.id, { onDelete: "cascade" })
    .notNull(),

  scanCount: integer("scan_count").default(1),

  lastScannedAt: timestamp("last_scanned_at", { withTimezone: true }).defaultNow(),

  createdAt: timestamp("created_at", { withTimezone: true }).defaultNow(),
});
```

# 🧠 4. Main Dashboard Page

## 📁 `app/dashboard/page.tsx`

```tsx id="dash3"
import { useEffect, useState } from "react";

export default function Dashboard() {
  const [profile, setProfile] = useState<any>(null);

  useEffect(() => {
    fetch("/api/v2/dashboard/profile", {
      headers: {
        Authorization: `Bearer ${localStorage.getItem("token")}`,
      },
    })
      .then((r) => r.json())
      .then(setProfile);
  }, []);

  return (
    <div className="min-h-screen p-10 bg-gradient-to-br from-stone-100 to-amber-100">
      <h1 className="text-3xl font-bold">GEN ERA IDENTITY</h1>

      {profile && (
        <div className="mt-6 glass-card p-6 rounded-xl">
          <p className="text-xl">Welcome, {profile.name}</p>
          <p className="text-sm text-gray-600">{profile.email}</p>
        </div>
      )}
    </div>
  );
}
```
