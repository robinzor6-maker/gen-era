# 🔐 1. Admin Guard (أهم جزء)

📍 `lib/auth/useRequireAdmin.ts`

```ts
import { useEffect } from "react";
import { useRouter } from "next/navigation";
import { useAuthStore } from "@/store/authStore";

export const useRequireAdmin = () => {
  const router = useRouter();
  const { user, token } = useAuthStore();

  useEffect(() => {
    if (!token || !user) {
      router.push("/login");
      return;
    }

    if (user.role !== "admin") {
      router.push("/unauthorized");
    }
  }, [user, token]);

  return { user };
};
```

# 📦 3. Products Admin Page

📍 `app/admin/products/page.tsx`

```tsx
"use client";

import { useEffect, useState } from "react";

export default function AdminProductsPage() {
  const [products, setProducts] = useState([]);

  useEffect(() => {
    fetch("/api/v2/products")
      .then(res => res.json())
      .then(setProducts);
  }, []);

  return (
    <div>
      <h1 className="text-2xl font-bold mb-4">Products</h1>

      <table className="w-full bg-white shadow">
        <thead>
          <tr>
            <th>Name</th>
            <th>Price</th>
            <th>Stock</th>
          </tr>
        </thead>

        <tbody>
          {products.map((p: any) => (
            <tr key={p.id}>
              <td>{p.name}</td>
              <td>{p.price}</td>
              <td>{p.stock}</td>
            </tr>
          ))}
        </tbody>
      </table>
    </div>
  );
}
```

# 📊 5. Inventory Page (Important)

📍 `app/admin/inventory/page.tsx`

```tsx
"use client";

import { useEffect, useState } from "react";

export default function InventoryPage() {
  const [products, setProducts] = useState([]);

  useEffect(() => {
    fetch("/api/v2/products")
      .then(res => res.json())
      .then(setProducts);
  }, []);

  const lowStock = products.filter((p: any) => p.stock < 5);

  return (
    <div>
      <h1 className="text-2xl font-bold mb-4">Inventory</h1>

      <div className="text-red-500 mb-4">
        Low Stock Items: {lowStock.length}
      </div>

      {lowStock.map((p: any) => (
        <div key={p.id} className="p-3 bg-red-50 mb-2">
          {p.name} — Only {p.stock} left
        </div>
      ))}
    </div>
  );
}
```
